```tsx
'use client';

import { ChevronDownIcon, ShapesIcon } from 'lucide-react';
import { useState } from 'react';
import { ActivityGrid, type ActivityCell, type ActivityLevel } from '@/components/panelui/activity-grid';

/** R reached, P partial, N none: thirty days, read left to right. */
const PERIODS: Record<string, string> = {
  'Last month': 'RPNNRNNNNR RRPRPRPRRR NRNNRRNRPN',
  'This month': 'RRRPRNRRRR PRRRNNRRPR RRNRRRPRRN',
  'Three months ago': 'NNPNNRNNNP NRNNPNNRNN PNNRNNNPNN',
};

const LEVELS: Record<string, ActivityLevel> = { R: 'reached', P: 'partial', N: 'none' };

function cellsOf(pattern: string): ActivityCell[] {
  return [...pattern.replaceAll(' ', '')].map((letter) => ({ level: LEVELS[letter] }));
}

export default function ActivityGridMindfulness() {
  const [period, setPeriod] = useState('Last month');

  return (
    // `dark` keeps the card dark on a light page, as the design has it.
    <div className="dark flex w-full max-w-xl flex-col gap-6 rounded-[36px] bg-card p-6 text-card-foreground">
      <div className="flex items-center gap-3">
        <span className="flex size-11 shrink-0 items-center justify-center rounded-full bg-foreground/10">
          <ShapesIcon className="size-5" aria-hidden="true" />
        </span>
        <h3 className="min-w-0 flex-1 truncate text-lg text-muted-foreground">
          Mindfulness Activity
        </h3>
        <label className="relative">
          <span className="sr-only">Period</span>
          <select
            value={period}
            onChange={(event) => setPeriod(event.target.value)}
            className="h-11 cursor-pointer appearance-none rounded-full bg-foreground/[0.06] pe-10 ps-4 text-sm text-muted-foreground outline-none focus-visible:ring-2 focus-visible:ring-ring"
          >
            {Object.keys(PERIODS).map((name) => (
              <option key={name}>{name}</option>
            ))}
          </select>
          <ChevronDownIcon
            className="pointer-events-none absolute end-3.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground"
            aria-hidden="true"
          />
        </label>
      </div>

      <ActivityGrid cells={cellsOf(PERIODS[period])} columns={10} label={`Mindfulness, ${period.toLowerCase()}`} />
      <ActivityGrid.Legend />
    </div>
  );
}
```

</ComponentPreview>

Use it to show how often something happened over a stretch of days: a habit, a goal, a streak. Each day is one pill, drawn filled when the goal was reached, with a dashed outline when it was partly reached, and faint when nothing happened.

It answers "how did the month go" at a glance. It does not show how much: a day that just reached the goal looks the same as one that doubled it. For amounts, use a chart.

## Usage

```tsx
import { ActivityGrid, type ActivityCell } from "@/components/panelui/activity-grid"
```

```tsx
const days: ActivityCell[] = [
  { level: 'reached', label: '1 June, 24 minutes' },
  { level: 'partial', label: '2 June, 12 minutes' },
  { level: 'none', label: '3 June, no practice' },
]

<ActivityGrid cells={days} columns={10} />
<ActivityGrid.Legend />
```

## Composition

```tsx
<ActivityGrid cells={…} />
<ActivityGrid.Legend />
```

- **`ActivityGrid.Legend`** — A key to the three kinds of pill. Pass `labels` to use your own words for them.

## Examples

### Days you can pick

Pass `onValueChange` and every pill becomes a button. Hold the selected index in `value`. The grid is one Tab stop, and the arrow keys move between days.

<ComponentPreview name="activity-grid/pick-a-day" height={320}>

```tsx
'use client';

import { useState } from 'react';
import { ActivityGrid, type ActivityCell } from '@/components/panelui/activity-grid';

/** Minutes of practice each day in June, against a goal of twenty. */
const MINUTES = [
  24, 12, 0, 0, 21, 0, 0, 0, 0, 30, 22, 26, 9, 20, 14, 25, 11, 23, 28, 20, 0, 31, 0, 0, 22, 27, 0,
  24, 8, 0,
];
const GOAL = 20;

const DAYS: ActivityCell[] = MINUTES.map((minutes, index) => ({
  level: minutes >= GOAL ? 'reached' : minutes > 0 ? 'partial' : 'none',
  label: `${index + 1} June, ${minutes} minutes`,
}));

export default function ActivityGridPickADay() {
  const [day, setDay] = useState<number | null>(13);
  const minutes = day === null ? null : MINUTES[day];

  return (
    <div className="dark flex w-full max-w-xl flex-col gap-5 rounded-[36px] bg-card p-6 text-card-foreground">
      <ActivityGrid cells={DAYS} columns={10} value={day} onValueChange={setDay} label="Practice in June" />
      <p className="text-sm text-muted-foreground" aria-live="polite">
        {day === null || minutes === null ? (
          'Pick a day'
        ) : (
          <>
            <span className="font-medium text-foreground">{day + 1} June</span>
            {minutes === 0
              ? ' · no practice'
              : ` · ${minutes} of ${GOAL} minutes${minutes >= GOAL ? ', goal reached' : ''}`}
          </>
        )}
      </p>
    </div>
  );
}
```

</ComponentPreview>

### A week to a row

`columns` is how many pills go in a row. Seven makes each row a week, and the legend takes your own words for the three states.

<ComponentPreview name="activity-grid/columns" height={360}>

```tsx
'use client';

import { ActivityGrid, type ActivityCell, type ActivityLevel } from '@/components/panelui/activity-grid';

/** Four weeks, Monday first. */
const WEEKS = 'RRPRRNN RRRRPNR PRRNRNN RRRRRPN';
const LEVELS: Record<string, ActivityLevel> = { R: 'reached', P: 'partial', N: 'none' };
const CELLS: ActivityCell[] = [...WEEKS.replaceAll(' ', '')].map((letter) => ({
  level: LEVELS[letter],
}));

export default function ActivityGridColumns() {
  return (
    <div className="dark flex w-full max-w-xs flex-col gap-4 rounded-[28px] bg-card p-5 text-card-foreground">
      <div className="grid grid-cols-7 gap-1.5 text-center text-xs text-muted-foreground sm:gap-2">
        {['M', 'T', 'W', 'T', 'F', 'S', 'S'].map((day, index) => (
          <span key={index}>{day}</span>
        ))}
      </div>
      <ActivityGrid cells={CELLS} columns={7} label="Workouts, last four weeks" />
      <ActivityGrid.Legend
        className="gap-x-4 text-xs"
        labels={{ reached: 'Done', partial: 'Short', none: 'Rest' }}
      />
    </div>
  );
}
```

## API Reference

### ActivityGrid

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `cells` | `ActivityCell[]` | — | The days, in order. They fill the grid row by row. |
| `columns` | `number` | `10` | Pills per row. |
| `value` | `number \| null` | `null` | The index of the selected day, when the caller holds it. |
| `onValueChange` | `(index: number) => void` | — | Called with the index of a day when it is pressed. Passing this makes the pills buttons; without it the grid is a picture. |
| `label` | `string` | `'Activity'` | What a screen reader calls the grid. |

### ActivityGrid.Legend

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `labels` | `Partial<Record<ActivityLevel, ReactNode>>` | — | Your own words for the three states. Any left out keep the default. |

Every part also takes the props of the element it renders, and a `className`.

## Notes

### The three states

`level` is `reached`, `partial` or `none`. Deciding which a day is, from minutes or steps or anything else, is up to you. The states are drawn differently as well as tinted differently, so the grid does not depend on colour alone.

### Labels

Give each cell a `label` such as "14 June, 22 minutes". It is the tooltip on hover and what a screen reader reads for the day. Without labels the grid is announced once, as a count of how many days reached the goal.

### Size

The grid fills its container and the pills keep their shape, so it gets taller as it gets wider. Put it in something with a width.

### Colour

The filled pill is the text colour with a little of the accent mixed in. Set `--panelui-accent` on any ancestor to change the accent for every PanelUI widget inside it.

### Server Components

`ActivityGrid` is a client component. In a file that starts with `"use client"`, write its parts with a dot, as on this page. A Server Component cannot read `ActivityGrid.Legend` off the import, so each part is also exported by name: `ActivityGridLegend` is `ActivityGrid.Legend`.

```tsx
import { ActivityGrid, ActivityGridLegend } from "@/components/panelui/activity-grid"
```
