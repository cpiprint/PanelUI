```tsx
'use client';

import { useEffect, useState } from 'react';
import { RollingNumber } from '@/components/panelui/rolling-number';

/** What a live count does over a minute: it wanders. */
const READINGS = [124, 121, 133, 129, 140, 136];

export default function RollingNumberCounter() {
  const [step, setStep] = useState(0);

  useEffect(() => {
    const timer = setInterval(() => {
      setStep((current) => (current + 1) % READINGS.length);
    }, 2200);
    return () => clearInterval(timer);
  }, []);

  return (
    <div className="flex flex-col gap-1">
      <span className="text-sm text-muted-foreground">Events</span>
      <RollingNumber
        value={READINGS[step]}
        className="text-6xl font-semibold tracking-tight"
      />
    </div>
  );
}
```

</ComponentPreview>

Use it for a figure that changes while someone is looking at it: a live count, a balance, a score. Each digit rolls to its new value on its own, so going from 124 to 121 moves one column and leaves the other two where they were.

The number is written by `Intl.NumberFormat`. Currency, percent and compact notation are options you pass, not strings you build.

For a number that is shown once and never changes, plain text does the same job.

## Examples

### Currency

`format` takes what `Intl.NumberFormat` takes. The currency sign and the separators stay where they are while the digits roll.

<ComponentPreview name="rolling-number/currency" height={280}>

```tsx
'use client';

import { useState } from 'react';
import { Button } from '@/components/ui/button';
import { RollingNumber } from '@/components/panelui/rolling-number';

export default function RollingNumberCurrency() {
  const [balance, setBalance] = useState(12582210.27);

  return (
    <div className="flex flex-col items-center gap-6">
      <div className="flex flex-col items-center gap-1">
        <span className="text-sm text-muted-foreground">Balance</span>
        <RollingNumber
          value={balance}
          format={{ style: 'currency', currency: 'USD' }}
          className="text-4xl font-semibold tracking-tight"
        />
      </div>
      <div className="flex flex-wrap justify-center gap-2">
        <Button
          variant="outline"
          size="sm"
          onClick={() => setBalance((value) => value - 1840.5)}
        >
          Pay an invoice
        </Button>
        <Button size="sm" onClick={() => setBalance((value) => value + 25000)}>
          Add a deposit
        </Button>
      </div>
    </div>
  );
}
```

### Percent and compact

Three figures in three notations, changed together. Compact notation keeps a large number short: 46,200 is written `$46.2K`.

<ComponentPreview name="rolling-number/formats" height={280}>

```tsx
'use client';

import { useState } from 'react';
import { Button } from '@/components/ui/button';
import { RollingNumber } from '@/components/panelui/rolling-number';

const WEEKS = [
  { visitors: 18420, conversion: 0.031, revenue: 46200 },
  { visitors: 21960, conversion: 0.038, revenue: 61850 },
  { visitors: 19310, conversion: 0.027, revenue: 39400 },
];

export default function RollingNumberFormats() {
  const [week, setWeek] = useState(0);
  const { visitors, conversion, revenue } = WEEKS[week];

  return (
    <div className="flex flex-col items-center gap-6">
      <dl className="grid grid-cols-3 gap-8 text-center">
        <div className="flex flex-col gap-1">
          <dt className="text-sm text-muted-foreground">Visitors</dt>
          <dd className="text-2xl font-semibold">
            <RollingNumber value={visitors} />
          </dd>
        </div>
        <div className="flex flex-col gap-1">
          <dt className="text-sm text-muted-foreground">Conversion</dt>
          <dd className="text-2xl font-semibold">
            <RollingNumber
              value={conversion}
              format={{ style: 'percent', minimumFractionDigits: 1 }}
            />
          </dd>
        </div>
        <div className="flex flex-col gap-1">
          <dt className="text-sm text-muted-foreground">Revenue</dt>
          <dd className="text-2xl font-semibold">
            <RollingNumber
              value={revenue}
              format={{
                style: 'currency',
                currency: 'USD',
                notation: 'compact',
                maximumFractionDigits: 1,
              }}
            />
          </dd>
        </div>
      </dl>
      <Button
        variant="outline"
        size="sm"
        onClick={() => setWeek((current) => (current + 1) % WEEKS.length)}
      >
        Next week
      </Button>
    </div>
  );
}
```

### A slower roll

`duration` is how long one digit takes to arrive, in milliseconds. It is 600 unless you set it.

<ComponentPreview name="rolling-number/slow" height={280}>

```tsx
'use client';

import { useState } from 'react';
import { Button } from '@/components/ui/button';
import { RollingNumber } from '@/components/panelui/rolling-number';

export default function RollingNumberSlow() {
  const [score, setScore] = useState(2048);

  return (
    <div className="flex flex-col items-center gap-6">
      <RollingNumber
        value={score}
        duration={1400}
        className="text-5xl font-semibold tracking-tight"
      />
      <Button variant="outline" size="sm" onClick={() => setScore((value) => value * 2)}>
        Double it
      </Button>
    </div>
  );
}
```

## API Reference

### RollingNumber

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `value` | `number` | — | The number to show. Each digit rolls to its new value on its own. |
| `format` | `Intl.NumberFormatOptions` | — | How the number is written: currency, percent, compact notation, how many decimals. Passed to `Intl.NumberFormat`. |
| `locales` | `Intl.LocalesArgument` | `'en-US'` | The locale the number is written in. Defaults to `en-US` rather than the browser's, so a page rendered on a server writes the same thing the browser then does. |
| `duration` | `number` | `600` | How long a digit takes to roll, in milliseconds. |

Every part also takes the props of the element it renders, and a `className`.

## Notes

### Width

The digits are set in tabular figures, so a 1 is as wide as an 8 and the number keeps its width while it rolls. It does get wider when it gains a digit, as 99 does on its way to 100. Give it room, or a fixed width, where something sits beside it.

### Server rendering

`locales` defaults to `en-US`, not to the browser's locale. A page rendered on a server would otherwise write the number one way and the browser another, and React reports that as a hydration mismatch. Pass the locale you want written.

### Reduced motion

With the system's reduce-motion setting on, a digit changes without rolling.

### Screen readers

A screen reader is given the formatted number once, as text. The rolling columns are hidden from it, so it does not read ten digits per column.
